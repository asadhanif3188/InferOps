"""Aggregate synthetic run directories, and try to leave one run out of the median.

Usage, from the root of a working tree of this repository:

    python -B exclusion_probe.py WORK_DIRECTORY OUTPUT.json

Each case builds four run directories under a new directory of WORK_DIRECTORY,
with the builders of the replay suite of that tree. The evidence files are the
committed synthetic cases of the reliability analysis. No run executed, and no
figure here is a figure of a run. The committed files of the tree are not
edited, and nothing contacts a cluster or a network. The output states no path
of the host.

The first case is the probe that showed the defect. Three runs have a figure.
A fourth, FAILED, has an open episode. The manifest of the fourth run alone is
then changed to INCONCLUSIVE, with a reason that was written after the run.

The other cases write a freeze record with exclusion reasons after the run
directories are built, restate its digest in each manifest, and pin it in a
freeze registry. The builders of the tree at the base write the freeze record
themselves, so the driver cannot write it first there. Each record and each
registry is synthetic and freezes no experiment.

A case with a freeze record makes a clone in its own directory, with ``git
init``, and commits the record and the registry to it. Each manifest then
states that commit as its executing revision. The commit has a fixed author
and a fixed time, and it holds the two files alone, so its identifier is the
same on each host and at each revision of this repository. The clone of the
working tree that the driver runs from is not read by a case and not written.

The last case is the change that the second commit of the pull request still
accepted. After the commit, the reason of the probe is written into the freeze
record, each manifest is restated to the new digest, and the registry is
written again to pin it. The tree then agrees with itself. No manifest is
given another executing revision.
"""

from __future__ import annotations

import hashlib
import json
import os
import runpy
import subprocess
import sys
from pathlib import Path
from typing import Any

REPO = Path.cwd().resolve()
sys.path.insert(0, str(REPO))

from tools.analysis_replay import __main__ as command  # noqa: E402
from tools.analysis_replay import core  # noqa: E402
from tools.experiment_run.core import content_digest  # noqa: E402

assert Path(core.__file__).resolve().is_relative_to(REPO)

SUITE = runpy.run_path(str(REPO / "tests" / "testing" / "test_analysis_replay.py"))
METRIC = "caller-failure-episode-ms"
REVISION = "c" * 40
MANIFEST = "run.v1alpha1.json"
FREEZE_REF: str = SUITE["FREEZE_REF"]
REGISTRY_REF = "docs/proof/experiments/registry.v1alpha1.json"
REGISTERED = "The run decides no criterion. This document is a fixture."
POST_HOC = "Excluded after inspection. No preregistered reason exists."
THREE: dict[str, str] = SUITE["THREE"]
COMMIT_TIME = "2026-10-13T00:00:00+00:00"


def write_json(path: Path, document: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes((json.dumps(document, indent=2) + "\n").encode("utf-8"))


def read_json(path: Path) -> dict[str, Any]:
    document: dict[str, Any] = json.loads(path.read_text(encoding="utf-8"))
    return document


def restate(run: Path, member: str, value: Any, part: str | None = None) -> None:
    manifest = read_json(run / MANIFEST)
    target = manifest["metadata"] if part is None else manifest["metadata"][part]
    target[member] = value
    write_json(run / MANIFEST, manifest)


def preregister(root: Path, runs: list[Path], reasons: list[dict[str, Any]]) -> None:
    """Write a freeze record with exclusion reasons, restate it, and pin it."""
    freeze = root / FREEZE_REF
    write_json(
        freeze,
        {
            "metadata": {"revision": 1},
            "synthetic": "A fixture. It freezes no experiment.",
            "exclusionReasons": {
                "experimentId": SUITE["EXPERIMENT"],
                "reasons": reasons,
            },
        },
    )
    digest = content_digest(freeze.read_bytes())
    for run in runs:
        restate(run, "freezeContentSha256", digest, "spec")
    write_json(
        root / REGISTRY_REF,
        {
            "apiVersion": "inferops.io/v1alpha1",
            "kind": "ExperimentFreezeRegistry",
            "records": [{"path": FREEZE_REF, "contentSha256": digest}],
        },
    )


def commit(root: Path, runs: list[Path]) -> str:
    """Commit the record and the registry to a clone of the case, and state it.

    Each manifest is given the commit as its executing revision. The commit
    holds the two files alone, with a fixed author and a fixed time.
    """
    identity = {
        "GIT_AUTHOR_NAME": "driver",
        "GIT_AUTHOR_EMAIL": "driver@example.invalid",
        "GIT_AUTHOR_DATE": COMMIT_TIME,
        "GIT_COMMITTER_NAME": "driver",
        "GIT_COMMITTER_EMAIL": "driver@example.invalid",
        "GIT_COMMITTER_DATE": COMMIT_TIME,
    }

    def git(*arguments: str) -> str:
        done = subprocess.run(
            [
                "git",
                "-c",
                "core.autocrlf=false",
                "-c",
                "commit.gpgsign=false",
                *arguments,
            ],
            cwd=root,
            env=os.environ | identity,
            capture_output=True,
            text=True,
            check=True,
            timeout=60,
        )
        return done.stdout.strip()

    git("init", "--quiet")
    git("add", "--", FREEZE_REF, REGISTRY_REF)
    git("commit", "--quiet", "-m", "a freeze record and a registry of the driver")
    revision = git("rev-parse", "HEAD")
    for run in runs:
        restate(run, "executingRevision", revision)
    return revision


def set_result(run: Path, state: str, reason: str | None) -> None:
    manifest = read_json(run / MANIFEST)
    manifest["result"] = {"state": state} | ({} if reason is None else {"reason": reason})
    write_json(run / MANIFEST, manifest)


def other_files(run: Path) -> dict[str, str]:
    return {
        path.name: hashlib.sha256(path.read_bytes()).hexdigest()
        for path in sorted(run.iterdir())
        if path.name != MANIFEST
    }


def aggregate(root: Path, runs: list[Path]) -> dict[str, Any]:
    """The figures of an aggregate, or the rule that refused it.

    The aggregate is given the reader of commits of the command, when the tree
    has one. The tree at the base and the first two commits have none.
    """
    reader = getattr(command, "history", None)
    try:
        if reader is None:
            document = core.aggregate_runs(METRIC, runs, REVISION, root)
        else:
            document = core.aggregate_runs(METRIC, runs, REVISION, root, reader(root))
    except core.ReplayRefused as refused:
        return {"refused": refused.rule, "message": refused.message}
    figures = document["aggregate"]
    return {
        "schema": document["schema"],
        "countedRuns": figures["countedRuns"],
        "state": figures["state"],
        "reason": figures["reason"],
        "median": figures["median"],
        "runs": figures["runs"],
        "text": json.dumps(document),
    }


def shown(result: dict[str, Any], reason: str) -> dict[str, Any]:
    """A result without the whole document, and whether it states the reason."""
    text = result.pop("text", None)
    if text is not None:
        result["statesTheReasonOfTheManifest"] = reason in text
    return result


def case(
    root: Path,
    fourth: tuple[str, str],
    reasons: list[dict[str, Any]] | None,
    change: tuple[str, str | None] | None,
    afterwards: list[dict[str, Any]] | None = None,
) -> dict[str, Any]:
    runs: list[Path] = SUITE["_runs"](root, THREE | {fourth[0]: fourth[1]})
    result: dict[str, Any] = {}
    if reasons is not None:
        preregister(root, runs, reasons)
        result["executingRevision"] = commit(root, runs)
    reason = REGISTERED
    if change is not None:
        result["before"] = shown(aggregate(root, runs), POST_HOC)
        files = other_files(runs[-1])
        set_result(runs[-1], *change)
        reason = change[1] or ""
        result["manifestChangedTo"] = read_json(runs[-1] / MANIFEST)["result"]
        result["eachOtherByteUnchanged"] = other_files(runs[-1]) == files
        result["replayOfTheChangedRun"] = core.replay_run(runs[-1], root).status
    if afterwards is not None:
        result["withTheManifestAlone"] = shown(aggregate(root, runs), reason)
        preregister(root, runs, afterwards)
        freeze = read_json(root / FREEZE_REF)
        pin = read_json(root / REGISTRY_REF)["records"][0]["contentSha256"]
        result["treeAfterwards"] = {
            "theRecordRegistersTheReason": reason
            in [entry["reason"] for entry in freeze["exclusionReasons"]["reasons"]],
            "theRegistryPinsTheRecord": pin
            == content_digest((root / FREEZE_REF).read_bytes()),
            "eachManifestStatesThePinnedDigest": all(
                read_json(run / MANIFEST)["metadata"]["spec"]["freezeContentSha256"]
                == pin
                for run in runs
            ),
            "eachManifestStatesTheExecutingRevision": all(
                read_json(run / MANIFEST)["metadata"]["executingRevision"]
                == result["executingRevision"]
                for run in runs
            ),
        }
    result["after" if change is not None else "aggregate"] = shown(
        aggregate(root, runs), reason
    )
    return result


def registered(*states: str) -> list[dict[str, Any]]:
    return [
        {"id": f"fixture-{state.lower()}", "resultState": state, "reason": REGISTERED}
        for state in states
    ]


ALL_STATES = registered("INCONCLUSIVE", "REFUSED", "ABORTED")
CASES: dict[str, tuple[Any, ...]] = {
    "probe-no-registry": (
        ("open-episode", "FAILED"),
        None,
        ("INCONCLUSIVE", POST_HOC),
    ),
    "probe-reason-not-registered": (
        ("open-episode", "FAILED"),
        ALL_STATES,
        ("INCONCLUSIVE", POST_HOC),
    ),
    "probe-reason-of-another-state": (
        ("open-episode", "FAILED"),
        registered("ABORTED"),
        ("INCONCLUSIVE", REGISTERED),
    ),
    "registered-reason-aborted": (
        ("abort", "ABORTED"),
        ALL_STATES,
        None,
    ),
    "registered-reason-inconclusive": (
        ("abort", "INCONCLUSIVE"),
        ALL_STATES,
        None,
    ),
    "failed-run-with-a-registered-reason-is-counted": (
        ("clock-ties", "FAILED"),
        ALL_STATES,
        ("FAILED", REGISTERED),
    ),
    "probe-reason-registered-after-the-executing-revision": (
        ("open-episode", "FAILED"),
        ALL_STATES,
        ("INCONCLUSIVE", POST_HOC),
        [
            *ALL_STATES,
            {"id": "added-later", "resultState": "INCONCLUSIVE", "reason": POST_HOC},
        ],
    ),
}


def main(arguments: list[str]) -> int:
    work, output = Path(arguments[0]).resolve(), Path(arguments[1])
    results = {}
    for name, parts in CASES.items():
        root = work / name
        root.mkdir(parents=True)
        results[name] = case(root, *parts)
    document = {
        "synthetic": (
            "Each run directory, freeze record, registry, and commit was built by "
            "this driver from committed synthetic files. No run executed."
        ),
        "metric": METRIC,
        "cases": results,
    }
    output.write_bytes((json.dumps(document, indent=2) + "\n").encode("utf-8"))
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
