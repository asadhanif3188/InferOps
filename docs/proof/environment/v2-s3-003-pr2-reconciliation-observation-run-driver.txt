#!/usr/bin/env bash
# Drives one reconciliation observation on the docker-desktop provider and prints
# one transcript. Run from the repository root, with a run name as the one
# argument. kubectl must be on PATH within one minor version of the server. The
# API image and the model seed image must be built on the host already.
#
# It prepares the cluster, applies the Application, observes it four times with
# the procedure's `observe` operation, and returns the cluster to the state it
# found. It makes no manual change to an object that the Application manages.
#
# When a preparation step ends the driver, the driver cleans up nothing. What
# it prepared stays: the images in the node, the prerequisite layer, and, by the
# step that failed, a Helm release or Argo CD. The message names the procedures
# that remove them.
#
# A preparation step that fails ends the driver. The Application's own values
# select the download source, so an apply over a claim that the preparation did
# not fill would download the model. The first version of this driver did not
# stop, and attempt 1 reached the apply after its preparation had failed.
#
# The throwaway Helm release raises the memory limit of the acquisition job. In
# attempt 1, on a Kubernetes v1.36.1 node, that job was stopped as out of memory
# at the chart's limit of 128Mi. The override applies to the throwaway release
# only. The Application sets no such value, and its acquisition job runs at the
# chart's limit.
#
# The four observations:
#   absent   before the Application exists
#   apply    while the procedure applies the Application
#   steady   after the apply returned
#   remove   while the procedure removes the Application
#
# After each observation the evidence tool builds one record from the directory
# that the observation wrote. The record is written under .artifacts/, and this
# driver prints its summary and its transitions.
set -u

RUN="${1:?give a run name: lowercase letters, digits, and hyphens}"
SCRIPT="scripts/environment/argocd-application.sh"
BOOTSTRAP="scripts/environment/argocd-bootstrap.sh"
PREREQUISITES="scripts/environment/terraform-prerequisites.sh"
NAMESPACE="inferops-release"
APPLICATION="local-docker-desktop-support-assistant"
API_SERVICE="inferops-inferops-llm"
API_DEPLOYMENT="inferops-inferops-llm"
KINDS="deployments,replicasets,pods,jobs,services,configmaps,serviceaccounts,networkpolicies,roles,rolebindings"
OBSERVATIONS=".artifacts/argocd-application/observations"

step() { printf '\n##### %s\n' "$*"; }

# Runs one command, prints it, and prints its exit status.
run() {
  printf '$ %s\n' "$*"
  "$@"
  printf '[exit %s]\n' "$?"
}

# Runs one preparation command. A failure ends the driver.
prepare() {
  printf '$ %s\n' "$*"
  "$@"
  local status=$?
  printf '[exit %s]\n' "${status}"
  [ "${status}" -eq 0 ] || abort "a preparation step failed"
}

# Ends the driver. Nothing is cleaned up here: the operator reads the state
# first.
abort() {
  printf 'ABORTED: %s. Nothing after it was run, and nothing was cleaned up.\n' "$1"
  printf 'To return the cluster to the state it had, run what applies, in this order:\n'
  printf '  helm --kube-context docker-desktop uninstall inferops --namespace %s\n' "${NAMESPACE}"
  printf '  bash %s remove --confirm\n' "${SCRIPT}"
  printf '  bash %s remove --confirm\n' "${BOOTSTRAP}"
  printf '  bash %s destroy --confirm\n' "${PREREQUISITES}"
  exit 1
}

k() { kubectl --context docker-desktop "$@"; }

# Builds the record of one observation and prints what it counts. The record
# holds the samples. This prints the summary, the transitions, and, for each
# sample, the read state, the five required fields, and whether it is settled.
record() {
  local name="${RUN}-$1" file
  file="${OBSERVATIONS}/${name}.record.json"
  printf '$ python -m tools.reconciliation_evidence %s/%s > %s\n' "${OBSERVATIONS}" "${name}" "${file}"
  uv run --locked python -m tools.reconciliation_evidence "${OBSERVATIONS}/${name}" >"${file}"
  printf '[exit %s]\n' "$?"
  run sha256sum "${file}"
  uv run --locked python - "${file}" <<'PYTHON'
import json
import sys

record = json.load(open(sys.argv[1], encoding="utf-8"))
print("collection complete:", record["collection"]["complete"])
print("summary:", json.dumps(record["summary"], sort_keys=True))
print("distinct lists of objects:", len(record["objectSets"]))


def shown(field):
    return field.get("value", "<" + field["state"] + ">")


for sample in record["samples"]:
    fields = sample["fields"]
    print(
        f"sample {sample['index']:03d} {shown(sample['observedAt'])}"
        f" read={sample['applicationRead']}"
        f" sync={shown(fields['syncStatus'])}"
        f" health={shown(fields['healthStatus'])}"
        f" phase={shown(fields['operationPhase'])}"
        f" resolved={str(shown(fields['resolvedRevision']))[:12]}"
        f" operation={str(shown(fields['operationRevision']))[:12]}"
        f" settled={sample['settled']['value']}"
        f" objects={sample['objectsRead']}:{sample['objectCount']}"
        f" consistency={sample['consistency']['state']}"
        f"/{sample['consistency'].get('revisions', {}).get('state', '-')}"
        f"/{sample['consistency'].get('source', {}).get('state', '-')}"
        f"/{sample['consistency'].get('labels', {}).get('state', '-')}"
    )
for transition in record["transitions"]:
    print(
        f"transition {transition['field']}"
        f" {transition['fromSample']:03d}->{transition['toSample']:03d}"
        f" {shown(transition['from'])} -> {shown(transition['to'])}"
    )
# One line for each of the three comparisons: how many samples made it, and
# how many findings they hold. A comparison that was not made is counted apart.
findings = []
for kind in ("revisions", "source", "labels"):
    sections = [
        sample["consistency"][kind]
        for sample in record["samples"]
        if sample["consistency"]["state"] == "provenance-resolved"
    ]
    states = sorted({section["state"] for section in sections})
    made = [section for section in sections if section["state"] != "not-compared"]
    count = sum(len(section["findings"]) for section in made)
    print(
        f"comparison {kind}: made in {len(made)} sample(s), not made in"
        f" {len(sections) - len(made)}, findings {count}, states {states}"
    )
for sample in record["samples"]:
    if sample["consistency"]["state"] != "provenance-resolved":
        continue
    for kind in ("revisions", "source", "labels"):
        for finding in sample["consistency"][kind].get("findings", []):
            findings.append((sample["index"], kind, finding))
print("consistency findings:", len(findings))
for index, kind, finding in findings[:40]:
    print(f"finding sample {index:03d} {kind} {finding['rule']} {finding['subject']}")
PYTHON
}

# One request that a caller sends, through a port-forward to the API Service.
# Nothing here reads Argo CD. kubectl is started directly, so that killing the
# job stops the forward.
caller_request() {
  local port="$1" forward
  kubectl --context docker-desktop port-forward "service/${API_SERVICE}" "${port}:8090" \
    -n "${NAMESPACE}" >.artifacts/reconciliation-observation-forward.log 2>&1 &
  forward=$!
  sleep 5
  run cat .artifacts/reconciliation-observation-forward.log
  run curl --silent --show-error --max-time 300 \
    --write-out '\nhttp_status=%{http_code} seconds=%{time_total}\n' \
    --header 'Content-Type: application/json' \
    --data '{"model":"qwen3-1-7b-q8-0","messages":[{"role":"user","content":"Reply with the single word: ready"}]}' \
    "http://127.0.0.1:${port}/v1/chat/completions"
  kill "${forward}" 2>/dev/null
  wait "${forward}" 2>/dev/null
}

step "0. Tools, revision, and the cluster before the run"
run date -u +%Y-%m-%dT%H:%M:%SZ
run git rev-parse HEAD
run git status --short
run git ls-remote origin refs/heads/main
run sha256sum "${SCRIPT}" "${BOOTSTRAP}" scripts/environment/lib.sh \
  tools/reconciliation_evidence/core.py tools/reconciliation_evidence/__main__.py \
  infra/argocd/workloads-project.yaml "infra/argocd/${APPLICATION}.yaml"
run kubectl version --context docker-desktop
run k get nodes -o name
run k get namespaces -o name

step "1. observe with no provider selected"
unset INFEROPS_PROVIDER INFEROPS_KIND_CLUSTER_NAME
run bash "${SCRIPT}" observe --samples 1 --interval 1 --into "${RUN}-refused"

export INFEROPS_PROVIDER=docker-desktop

step "2. observe with no bound"
run bash "${SCRIPT}" observe --into "${RUN}-refused"

step "3. observe before Argo CD is installed"
run bash "${SCRIPT}" observe --samples 2 --interval 1 --into "${RUN}-no-controller"
record no-controller

step "4. Preparation: the two images, loaded into the node"
prepare bash scripts/environment/api-image.sh load
prepare bash scripts/environment/model-seed-image.sh load

step "5. Preparation: the prerequisite layer"
prepare bash "${PREREQUISITES}" apply

step "6. Preparation: fill the claim from the local seed image, with one throwaway Helm release"
API_IMAGE_DIGEST="$(bash scripts/environment/api-image.sh digest)" ||
  abort "api-image.sh digest failed"
printf '[exit 0] api-image.sh digest\n'
printf 'API image digest on this host: %s\n' "${API_IMAGE_DIGEST}"
bash scripts/environment/api-image.sh values >.artifacts/api-image-values.yaml ||
  abort "api-image.sh values failed"
printf '[exit 0] api-image.sh values\n'
bash scripts/environment/model-seed-image.sh values >.artifacts/model-seed-values.yaml ||
  abort "model-seed-image.sh values failed"
printf '[exit 0] model-seed-image.sh values\n'
prepare helm --kube-context docker-desktop install inferops charts/inferops-llm \
  --namespace "${NAMESPACE}" \
  --values charts/inferops-llm/ci/real-values.yaml \
  --values .artifacts/api-image-values.yaml \
  --values .artifacts/model-seed-values.yaml \
  --set model.acquisition.resources.limits.memory=2Gi \
  --wait --timeout 15m
prepare helm --kube-context docker-desktop uninstall inferops --namespace "${NAMESPACE}" --wait --timeout 600s
# An object that the throwaway release left would be met by the Application's
# sync. In attempt 1 a failed acquisition job was left, and the sync waited on
# it. A query that does not answer is not read as no object.
left="unread"
for _ in $(seq 1 120); do
  left="$(k get "${KINDS}" -n "${NAMESPACE}" -l app.kubernetes.io/instance=inferops -o name)" ||
    abort "the query for what the throwaway release left did not answer"
  [ -n "${left}" ] || break
  sleep 2
done
printf 'What the throwaway release left: %s\n' "${left:-nothing}"
[ -z "${left}" ] || abort "the throwaway release left an object"

step "7. Preparation: install Argo CD"
prepare bash "${BOOTSTRAP}" install

step "8. observe: no Application exists"
run bash "${SCRIPT}" observe --samples 3 --interval 2 --into "${RUN}-absent"
record absent

step "9. observe into a directory that exists"
run bash "${SCRIPT}" observe --samples 3 --interval 2 --into "${RUN}-absent"

step "10. observe, while the procedure applies the Application"
bash "${SCRIPT}" observe --samples 80 --interval 3 --into "${RUN}-apply" \
  >.artifacts/reconciliation-observation-apply.log 2>&1 &
observer=$!
sleep 15
run bash "${SCRIPT}" apply --api-image-digest "${API_IMAGE_DIGEST}"
wait "${observer}"
printf '[exit %s] observe, in the background\n' "$?"
run cat .artifacts/reconciliation-observation-apply.log
record apply

step "11. verify"
run bash "${SCRIPT}" verify
run k rollout status "deployment/${API_DEPLOYMENT}" -n "${NAMESPACE}" --timeout=600s
run k rollout status "deployment/${API_DEPLOYMENT}-runtime" -n "${NAMESPACE}" --timeout=600s

step "12. observe: after the apply returned"
run bash "${SCRIPT}" observe --samples 5 --interval 2 --into "${RUN}-steady"
record steady
# The labels of the applied objects, asked with plain kubectl and not through
# the procedure.
run k get "${KINDS}" -n "${NAMESPACE}" -l app.kubernetes.io/instance=inferops \
  -o 'jsonpath={range .items[*]}{.kind}/{.metadata.name} chart={.metadata.labels.helm\.sh/chart} version={.metadata.labels.app\.kubernetes\.io/version} workload={.metadata.labels.inferops\.io/workload}{"\n"}{end}'

step "13. One caller request. Nothing here reads Argo CD"
caller_request 18093

step "14. observe, while the procedure removes the Application"
bash "${SCRIPT}" observe --samples 40 --interval 2 --into "${RUN}-remove" \
  >.artifacts/reconciliation-observation-remove.log 2>&1 &
observer=$!
sleep 10
run bash "${SCRIPT}" remove --confirm
wait "${observer}"
printf '[exit %s] observe, in the background\n' "$?"
run cat .artifacts/reconciliation-observation-remove.log
record remove

step "15. Cleanup: remove Argo CD and the prerequisite layer"
run bash "${BOOTSTRAP}" remove --confirm
run bash "${PREREQUISITES}" destroy --confirm

step "16. The cluster after the run, asked independently"
run k get namespaces -o name
run k get customresourcedefinitions -o name
run git status --short
run date -u +%Y-%m-%dT%H:%M:%SZ
