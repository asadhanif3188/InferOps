#!/usr/bin/env bash
# Drives one observation of the model cache under two serving runtime replicas,
# on the docker-desktop provider, and prints one transcript. Run it from the
# repository root:
#
#   bash driver.sh up RUN        prepare the cluster, read capacity, apply the release
#   bash driver.sh collect RUN   read the pods and build the record
#   bash driver.sh down RUN      remove what `up` installed
#
# RUN is a short name such as run-1. The three phases are three invocations, so
# that a collection that stopped on a defect of this driver can be made again
# while the same pods run. It was copied to a directory that Git ignores before
# it ran, so the working tree was clean in step P1.
#
# It is written to download no model: one preparation Helm release fills the
# model cache claim from the local seed image, and is uninstalled before the
# Application is applied. The API image and the model seed image are built from
# the local build cache and loaded into the node.
#
# What it reads, and how:
#   - `up` and `down` change the cluster through the repository's own
#     procedures. Each of them verifies the target before it writes.
#   - `collect` sends read requests, with three exceptions that change nothing
#     the release declares: it reads each runtime container's mount table and
#     the artifact's file identity with `kubectl exec`, and it sends each
#     runtime pod one model listing and one completion through a port-forward
#     to that pod. No response body is kept.
#
# Outcome rules that it follows:
#   - A precondition or a preparation step that fails ends `up` before the
#     Application is applied. Nothing is cleaned up: the operator reads the
#     cluster first, and then runs `down`.
#   - The capacity preflight is the V1 multi-replica preflight, with its own
#     measuring program and its own figures. When it does not exit 0, `up`
#     writes that outcome and ends. It does not apply the Application, and it
#     changes no replica count, request, or limit.
#   - From the apply on, nothing ends `up` early. A step that fails is printed
#     with its exit status.
#   - Each completion is sent once, with no retry.
#   - It computes no verdict. The tool that builds the record states the result.
set -u

PHASE="${1:?give the phase: up, collect, or down}"
RUN="${2:?give the run name, for example run-1}"
EVIDENCE="docs/proof/environment/v2-s4-002-pr2-runtime-model-cache-${RUN}"
WORK=".artifacts/runtime-model-cache/${RUN}"
APPLICATION_SCRIPT="scripts/environment/argocd-application.sh"
BOOTSTRAP="scripts/environment/argocd-bootstrap.sh"
PREREQUISITES="scripts/environment/terraform-prerequisites.sh"
V1_PREFLIGHT_SCRIPT="scripts/environment/kubernetes-multi-replica-certification.sh"
V1_DESCRIPTOR="deploy/serving/certification/k8s-multi-replica-inference.v1.json"
V1_CAPACITY_FACTS=".artifacts/kubernetes-multi-replica-certification/capacity-facts.json"
NAMESPACE="inferops-release"
CONTROLLER_NAMESPACE="argocd"
APPLICATION="local-docker-desktop-support-assistant"
API_DEPLOYMENT="inferops-inferops-llm"
RUNTIME_DEPLOYMENT="inferops-inferops-llm-runtime"
RUNTIME_SELECTOR="app.kubernetes.io/instance=inferops,app.kubernetes.io/component=serving-runtime"
RELEASE_DIRECTORY="gitops/environments/local-docker-desktop/workloads/support-assistant"
KINDS="deployments,replicasets,pods,jobs,services,configmaps,serviceaccounts,networkpolicies,roles,rolebindings"
RUNTIME_PORT=8080
FIRST_LOCAL_PORT=18220
ROLLOUT_BUDGET_SECONDS=900
SAMPLE_INTERVAL_SECONDS=5

# A path that starts with a slash is an argument for a container here, and not
# a path of this host.
export MSYS_NO_PATHCONV=1
export MSYS2_ARG_CONV_EXCL='*'

step() { printf '\n##### %s\n' "$*"; }
now() { date -u +%Y-%m-%dT%H:%M:%SZ; }

# Runs one command, prints it, and prints its exit status. The status is also
# left in LAST.
run() {
  printf '$ %s\n' "$*"
  "$@"
  LAST=$?
  printf '[exit %s]\n' "${LAST}"
}

refuse() {
  printf '\nREFUSED: %s. Nothing after it was run, and nothing was cleaned up.\n' "$1"
  printf 'The operator reads the state of the cluster first, and then runs: bash driver.sh down %s\n' "${RUN}"
  run now
  exit 1
}

# Runs one precondition or preparation command. A failure ends the phase.
prepare() {
  run "$@"
  [ "${LAST}" -eq 0 ] || refuse "a precondition or a preparation step failed: $*"
}

k() { kubectl --context docker-desktop "$@"; }

# Writes the output of one read to a file, with LF line ends, and prints the
# command and its exit status. The file is not written when the read fails.
read_into() {
  local target="$1"
  shift
  printf '$ %s > %s\n' "$*" "${target}"
  if "$@" >"${WORK}/read.tmp" 2>"${WORK}/read.err"; then
    tr -d '\r' <"${WORK}/read.tmp" >"${target}"
    LAST=0
  else
    LAST=$?
    cat "${WORK}/read.err"
  fi
  printf '[exit %s]\n' "${LAST}"
}

write_header() {
  # $1 is the time `collect` ended, or an empty text.
  printf '{\n  "schema": "inferops.io/runtime-model-cache-collection/v1alpha1",\n  "runId": "%s",\n  "provider": "docker-desktop",\n  "namespace": "%s",\n  "executingCommit": "%s",\n  "reportedRevision": "%s",\n  "startedAt": "%s",\n  "collectedAt": "%s"\n}\n' \
    "${RUN}" "${NAMESPACE}" "$(cat "${WORK}/executing-commit")" \
    "$(cat "${WORK}/reported-revision" 2>/dev/null)" "$(cat "${WORK}/started-at")" "$1" \
    >"${EVIDENCE}/run.json"
}

independent_state() {
  run k get namespaces -o name
  run k get customresourcedefinitions -o name
  run k get "${KINDS}" -n "${NAMESPACE}" -l app.kubernetes.io/instance=inferops -o name
  run k get persistentvolumeclaims -n "${NAMESPACE}" -o name
}

# --------------------------------------------------------------------------
up() {
  step "0. Start: the date and the tool versions"
  run now
  run kubectl version --client -o json
  run helm version --short
  run terraform version
  run docker version --format '{{.Client.Version}} {{.Server.Version}}'
  run uv --version
  run uv run --locked python --version
  run git --version

  step "P1. The executing commit, the working tree, and the commit the controller will read"
  prepare git rev-parse HEAD
  HEAD_COMMIT="$(git rev-parse HEAD)"
  prepare git status --porcelain --untracked-files=all
  [ -z "$(git status --porcelain --untracked-files=all)" ] || refuse "the working tree is not clean"
  prepare git ls-remote origin refs/heads/main
  REMOTE_COMMIT="$(git ls-remote origin refs/heads/main | awk '{ print $1 }')"
  printf 'executing commit: %s\nremote main:      %s\n' "${HEAD_COMMIT}" "${REMOTE_COMMIT}"
  # The controller reads the chart and the release from the remote main. The
  # tool and this driver are read from the executing commit. The two commits may
  # differ, and then the files the controller reads must not.
  prepare git diff --stat --exit-code "${REMOTE_COMMIT}" HEAD -- charts gitops infra/argocd
  [ ! -e "${EVIDENCE}" ] || refuse "the evidence directory exists"
  [ ! -e "${WORK}/started-at" ] || refuse "the work directory of this run name was used before"
  mkdir -p "${WORK}" "${EVIDENCE}"
  now >"${WORK}/started-at"
  printf '%s' "${HEAD_COMMIT}" >"${WORK}/executing-commit"

  step "P2. The files this run reads, and the identity the release declares"
  run sha256sum "$0" tools/runtime_model_cache/core.py tools/runtime_model_cache/__main__.py \
    tools/runtime_model_cache/__init__.py "${APPLICATION_SCRIPT}" "${BOOTSTRAP}" "${PREREQUISITES}" \
    "${V1_PREFLIGHT_SCRIPT}" "${V1_DESCRIPTOR}" scripts/environment/lib.sh \
    "${RELEASE_DIRECTORY}/values.generated.yaml" "${RELEASE_DIRECTORY}/rendered-workload-release.yaml"
  prepare uv run --locked python -m tools.gitops_desired_state --check
  read_into "${EVIDENCE}/expected.json" uv run --locked python -m tools.runtime_model_cache --expected
  [ "${LAST}" -eq 0 ] || refuse "the expected identity was not derived"
  run cat "${EVIDENCE}/expected.json"

  step "P3. The provider, and the cluster before the run"
  printf '$ export INFEROPS_PROVIDER=docker-desktop\n'
  export INFEROPS_PROVIDER=docker-desktop
  unset INFEROPS_KIND_CLUSTER_NAME
  prepare kubectl version --context docker-desktop
  prepare k get nodes -o wide
  read_into "${WORK}/namespaces.before" k get namespaces -o name
  [ "${LAST}" -eq 0 ] || refuse "the namespaces were not read"
  run cat "${WORK}/namespaces.before"
  read_into "${WORK}/crds.before" k get customresourcedefinitions -o name
  [ "${LAST}" -eq 0 ] || refuse "the custom resource definitions were not read"
  run cat "${WORK}/crds.before"
  prepare k get storageclasses
  prepare k get pods --all-namespaces
  if grep -q -e "^namespace/${NAMESPACE}\$" -e "^namespace/${CONTROLLER_NAMESPACE}\$" "${WORK}/namespaces.before"; then
    refuse "the cluster already holds the release namespace or the controller namespace"
  fi

  step "P4. The API image at the executing commit, and the model seed image"
  prepare bash scripts/environment/api-image.sh build
  prepare bash scripts/environment/api-image.sh load
  prepare bash scripts/environment/api-image.sh digest
  API_IMAGE_DIGEST="$(bash scripts/environment/api-image.sh digest)" || refuse "api-image.sh digest failed"
  printf '%s' "${API_IMAGE_DIGEST}" >"${WORK}/api-image-digest"
  prepare bash scripts/environment/model-seed-image.sh build
  prepare bash scripts/environment/model-seed-image.sh load
  prepare bash scripts/environment/model-seed-image.sh digest

  step "P5. The platform namespace and the model cache claim"
  prepare bash "${PREREQUISITES}" apply
  run k get persistentvolumeclaims -n "${NAMESPACE}" -o wide

  step "P6. Fill the claim from the seed image with one preparation Helm release, and remove that release"
  printf '$ bash scripts/environment/api-image.sh values > %s/api-image-values.yaml\n' "${WORK}"
  bash scripts/environment/api-image.sh values >"${WORK}/api-image-values.yaml" || refuse "api-image.sh values failed"
  printf '$ bash scripts/environment/model-seed-image.sh values > %s/model-seed-values.yaml\n' "${WORK}"
  bash scripts/environment/model-seed-image.sh values >"${WORK}/model-seed-values.yaml" || refuse "model-seed-image.sh values failed"
  # The preparation release is the V1 real values file: one API replica and one
  # runtime replica. It is not the release this run observes.
  prepare helm --kube-context docker-desktop install inferops charts/inferops-llm \
    --namespace "${NAMESPACE}" \
    --values charts/inferops-llm/ci/real-values.yaml \
    --values "${WORK}/api-image-values.yaml" \
    --values "${WORK}/model-seed-values.yaml" \
    --set model.acquisition.resources.limits.memory=2Gi \
    --wait --timeout 15m
  prepare helm --kube-context docker-desktop uninstall inferops --namespace "${NAMESPACE}" --wait --timeout 600s
  printf '$ kubectl --context docker-desktop get %s -n %s -l app.kubernetes.io/instance=inferops -o name\n' "${KINDS}" "${NAMESPACE}"
  printf '  repeated every 2 seconds, at most 120 times, until it prints nothing\n'
  left="unread"
  for _ in $(seq 1 120); do
    left="$(k get "${KINDS}" -n "${NAMESPACE}" -l app.kubernetes.io/instance=inferops -o name)" ||
      refuse "the query for what the preparation release left did not answer"
    [ -n "${left}" ] || break
    sleep 2
  done
  printf 'What the preparation release left: %s\n' "${left:-nothing}"
  [ -z "${left}" ] || refuse "the preparation release left an object"

  step "P7. Install the GitOps controller with the bootstrap procedure, and verify it"
  prepare bash "${BOOTSTRAP}" install
  prepare bash "${BOOTSTRAP}" verify

  step "C1. The V1 multi-replica capacity preflight, before the Application is applied"
  # The measuring program is the one the V1 script holds between its two
  # CAPACITY_PYTHON markers. It is taken from that script and not written again.
  awk "/<<'CAPACITY_PYTHON'/{f=1;next} /^CAPACITY_PYTHON\$/{f=0} f" "${V1_PREFLIGHT_SCRIPT}" >"${WORK}/capacity-program.py"
  run sha256sum "${WORK}/capacity-program.py"
  INFEROPS_CAPACITY_ENGINE_CPUS="$(docker info --format '{{.NCPU}}')" || refuse "the engine did not report its processors"
  INFEROPS_CAPACITY_ENGINE_MEMORY="$(docker info --format '{{.MemTotal}}')" || refuse "the engine did not report its memory"
  INFEROPS_CAPACITY_NODES="$(k get nodes -o json)" || refuse "the nodes were not read"
  INFEROPS_CAPACITY_PODS="$(k get pods --all-namespaces -o json)" || refuse "the pods were not read"
  export INFEROPS_CAPACITY_ENGINE_CPUS INFEROPS_CAPACITY_ENGINE_MEMORY INFEROPS_CAPACITY_NODES INFEROPS_CAPACITY_PODS
  mkdir -p "$(dirname "${V1_CAPACITY_FACTS}")"
  prepare uv run --locked python "${WORK}/capacity-program.py" "${V1_CAPACITY_FACTS}"
  unset INFEROPS_CAPACITY_NODES INFEROPS_CAPACITY_PODS
  tr -d '\r' <"${V1_CAPACITY_FACTS}" >"${EVIDENCE}/capacity-facts.json"
  run cat "${EVIDENCE}/capacity-facts.json"
  run uv run --locked python -m tools.kubernetes_certification.multi_replica_cli preflight
  PREFLIGHT_EXIT="${LAST}"
  printf '{\n  "exitStatus": %s\n}\n' "${PREFLIGHT_EXIT}" >"${EVIDENCE}/capacity-preflight.json"
  if [ "${PREFLIGHT_EXIT}" -ne 0 ]; then
    write_header ""
    printf '\nThe capacity preflight exited %s. The Application is not applied, and nothing is changed to fit.\n' "${PREFLIGHT_EXIT}"
    printf 'Run: bash driver.sh collect %s, and then: bash driver.sh down %s\n' "${RUN}" "${RUN}"
    run now
    exit 0
  fi

  step "D1. Apply the project and the Application, and sample the runtime pods while they start"
  : >"${EVIDENCE}/readiness-samples.txt"
  rm -f "${WORK}/stop-sampling"
  (
    # At most 480 samples, which is 40 minutes at the interval above.
    for _ in $(seq 1 480); do
      [ ! -e "${WORK}/stop-sampling" ] || break
      stamp="$(now)"
      k get pods -n "${NAMESPACE}" -l "${RUNTIME_SELECTOR}" \
        -o 'jsonpath={range .items[*]}{.metadata.name}{"\t"}{.status.phase}{"\t"}{range .status.initContainerStatuses[*]}{.state.terminated.exitCode}{end}{"\t"}{range .status.containerStatuses[*]}{.state.running.startedAt}{end}{"\t"}{range .status.containerStatuses[*]}{.started}{end}{"\t"}{range .status.containerStatuses[*]}{.ready}{end}{"\n"}{end}' \
        2>/dev/null | tr -d '\r' | while IFS= read -r line; do
        [ -z "${line}" ] || printf '%s\t%s\n' "${stamp}" "${line}"
      done >>"${EVIDENCE}/readiness-samples.txt"
      sleep "${SAMPLE_INTERVAL_SECONDS}"
    done
  ) &
  sampler=$!
  run now
  run bash "${APPLICATION_SCRIPT}" apply --api-image-digest "$(cat "${WORK}/api-image-digest")"

  step "D2. Wait for the two rollouts, with one bound each"
  run k rollout status "deployment/${API_DEPLOYMENT}" -n "${NAMESPACE}" --timeout="${ROLLOUT_BUDGET_SECONDS}s"
  run k rollout status "deployment/${RUNTIME_DEPLOYMENT}" -n "${NAMESPACE}" --timeout="${ROLLOUT_BUDGET_SECONDS}s"
  sleep "$((SAMPLE_INTERVAL_SECONDS * 2))"
  : >"${WORK}/stop-sampling"
  wait "${sampler}" 2>/dev/null
  run now
  run wc -l "${EVIDENCE}/readiness-samples.txt"

  step "D3. What the controller reports, and the commit it read"
  run bash "${APPLICATION_SCRIPT}" verify
  k get application "${APPLICATION}" -n "${CONTROLLER_NAMESPACE}" -o 'jsonpath={.status.sync.revision}' >"${WORK}/reported-revision"
  run k get application "${APPLICATION}" -n "${CONTROLLER_NAMESPACE}" \
    -o 'jsonpath=revision={.status.sync.revision} sync={.status.sync.status} health={.status.health.status} operation={.status.operationState.phase}{"\n"}'
  run git cat-file -t "$(cat "${WORK}/reported-revision")"
  run git diff --stat --exit-code "$(cat "${WORK}/reported-revision")" HEAD -- charts gitops infra/argocd
  write_header ""
  run k get pods -n "${NAMESPACE}" -o wide
  run now
}

# --------------------------------------------------------------------------
collect() {
  [ -e "${EVIDENCE}/expected.json" ] || { printf 'REFUSED: %s holds no expected.json. Run the phase up first.\n' "${EVIDENCE}"; exit 1; }
  mkdir -p "${WORK}"
  step "E0. Start of the collection"
  run now
  run sha256sum "$0" tools/runtime_model_cache/core.py
  run k get pods -n "${NAMESPACE}" -o wide
  run k get deployments -n "${NAMESPACE}" -o wide

  step "E1. The runtime pods, the claims, the volumes, and the events"
  read_into "${EVIDENCE}/runtime-pods.json" k get pods -n "${NAMESPACE}" -l "${RUNTIME_SELECTOR}" -o json
  read_into "${EVIDENCE}/claims.json" k get persistentvolumeclaims -n "${NAMESPACE}" -o json
  read_into "${EVIDENCE}/volumes.json" k get persistentvolumes -o json
  read_into "${EVIDENCE}/events.txt" k get events -n "${NAMESPACE}" --field-selector involvedObject.kind=Pod \
    -o 'jsonpath={range .items[*]}{.involvedObject.name}{"\t"}{.reason}{"\t"}{.count}{"\t"}{.firstTimestamp}{"\t"}{.lastTimestamp}{"\t"}{.message}{"\n"}{end}'
  run k get persistentvolumeclaims -n "${NAMESPACE}" -o wide
  run k get persistentvolumes -o wide

  step "E2. Each runtime pod: the verification log, the mount table, and the file identity"
  pods="$(k get pods -n "${NAMESPACE}" -l "${RUNTIME_SELECTOR}" -o 'jsonpath={range .items[*]}{.metadata.name}{"\n"}{end}' | tr -d '\r')"
  model_path="$(uv run --locked python -c 'import json,sys; print(json.load(open(sys.argv[1]))["containerPath"])' "${EVIDENCE}/expected.json" | tr -d '\r')"
  printf 'runtime pods:\n%s\nartifact path: %s\n' "${pods}" "${model_path}"
  for pod in ${pods}; do
    read_into "${EVIDENCE}/verify-model.${pod}.txt" k logs "${pod}" -n "${NAMESPACE}" -c verify-model
    run cat "${EVIDENCE}/verify-model.${pod}.txt"
    read_into "${EVIDENCE}/mountinfo.${pod}.txt" k exec "${pod}" -n "${NAMESPACE}" -c runtime -- cat /proc/self/mountinfo
    run grep -F " $(dirname "${model_path}") " "${EVIDENCE}/mountinfo.${pod}.txt"
    read_into "${EVIDENCE}/artifact.${pod}.txt" k exec "${pod}" -n "${NAMESPACE}" -c runtime -- stat -c '%d %i %s' "${model_path}"
    run cat "${EVIDENCE}/artifact.${pod}.txt"
  done

  step "E3. Each runtime pod: one model listing and one completion, through a port-forward to that pod"
  port="${FIRST_LOCAL_PORT}"
  for pod in ${pods}; do
    port=$((port + 1))
    alias_argument="$(k get pod "${pod}" -n "${NAMESPACE}" -o 'jsonpath={.spec.containers[0].args}' |
      uv run --locked python -c 'import json,sys; a=json.load(sys.stdin); print(a[a.index("--alias")+1])' | tr -d '\r')"
    printf '$ kubectl --context docker-desktop port-forward pod/%s %s:%s -n %s &\n' "${pod}" "${port}" "${RUNTIME_PORT}" "${NAMESPACE}"
    # kubectl is started directly. A background shell function is a subshell,
    # and killing it leaves kubectl listening on the port.
    kubectl --context docker-desktop port-forward "pod/${pod}" "${port}:${RUNTIME_PORT}" -n "${NAMESPACE}" >"${WORK}/forward.${pod}.txt" 2>&1 &
    forward=$!
    for _ in $(seq 1 30); do
      grep -q 'Forwarding from' "${WORK}/forward.${pod}.txt" 2>/dev/null && break
      sleep 1
    done
    run cat "${WORK}/forward.${pod}.txt"
    if grep -q 'Forwarding from' "${WORK}/forward.${pod}.txt" && kill -0 "${forward}" 2>/dev/null; then
      printf '$ curl GET http://127.0.0.1:%s/v1/models\n' "${port}"
      status="$(curl --silent --show-error --max-time 60 --output "${WORK}/models.${pod}.raw.json" --write-out '%{http_code}' "http://127.0.0.1:${port}/v1/models")"
      printf '[exit %s] http_status=%s\n' "$?" "${status}"
      uv run --locked python - "${WORK}/models.${pod}.raw.json" "${status}" "${EVIDENCE}/models.${pod}.json" <<'PYTHON'
import json
import sys
from pathlib import Path

raw, status, out = sys.argv[1], sys.argv[2], sys.argv[3]
try:
    listed = json.loads(Path(raw).read_text(encoding="utf-8"))["data"][0]
except (OSError, ValueError, KeyError, IndexError, TypeError):
    listed = {}
document = {
    "httpStatus": int(status) if status.isdigit() else None,
    "id": listed.get("id"),
    "meta": listed.get("meta"),
}
Path(out).write_text(json.dumps(document, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")
print(json.dumps(document, sort_keys=True))
PYTHON
      printf '$ curl POST http://127.0.0.1:%s/v1/chat/completions, model %s, max_tokens 16, once\n' "${port}" "${alias_argument}"
      status="$(curl --silent --show-error --max-time 300 --output "${WORK}/completion.${pod}.raw.json" --write-out '%{http_code} %{time_total}' \
        --header 'Content-Type: application/json' \
        --data "{\"model\":\"${alias_argument}\",\"messages\":[{\"role\":\"user\",\"content\":\"Reply with the single word: ready\"}],\"max_tokens\":16,\"stream\":false}" \
        "http://127.0.0.1:${port}/v1/chat/completions")"
      printf '[exit %s] http_status and seconds: %s\n' "$?" "${status}"
      uv run --locked python - "${WORK}/completion.${pod}.raw.json" "${status%% *}" "${EVIDENCE}/completion.${pod}.json" <<'PYTHON'
import json
import sys
from pathlib import Path

raw, status, out = sys.argv[1], sys.argv[2], sys.argv[3]
try:
    answer = json.loads(Path(raw).read_text(encoding="utf-8"))
except (OSError, ValueError):
    answer = {}
if not isinstance(answer, dict):
    answer = {}
choices = answer.get("choices") or [{}]
usage = answer.get("usage") or {}
document = {
    "httpStatus": int(status) if status.isdigit() else None,
    "model": answer.get("model"),
    "finishReason": choices[0].get("finish_reason"),
    "promptTokens": usage.get("prompt_tokens"),
    "completionTokens": usage.get("completion_tokens"),
}
Path(out).write_text(json.dumps(document, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")
print(json.dumps(document, sort_keys=True))
PYTHON
    else
      printf 'The port-forward to %s did not listen. No request was sent to that pod.\n' "${pod}"
    fi
    kill "${forward}" 2>/dev/null
    wait "${forward}" 2>/dev/null
  done

  step "E4. The state after the reads, and the record"
  run k get pods -n "${NAMESPACE}" -l "${RUNTIME_SELECTOR}" \
    -o 'jsonpath={range .items[*]}{.metadata.name} node={.spec.nodeName} restarts={.status.containerStatuses[0].restartCount} ready={.status.containerStatuses[0].ready}{"\n"}{end}'
  write_header "$(now)"
  run cat "${EVIDENCE}/run.json"
  read_into "${EVIDENCE}/record.v1alpha1.json" uv run --locked python -m tools.runtime_model_cache "${EVIDENCE}"
  printf '$ print the result and the state of each rule of %s/record.v1alpha1.json\n' "${EVIDENCE}"
  uv run --locked python - "${EVIDENCE}/record.v1alpha1.json" <<'PYTHON'
import json
import sys
from pathlib import Path

record = json.loads(Path(sys.argv[1]).read_text(encoding="utf-8"))
print("result:", record["result"])
for rule in record["rules"]:
    print(rule["state"].ljust(13), rule["id"] + ":", rule["detail"])
PYTHON
  printf '[exit %s]\n' "$?"
  run now
}

# --------------------------------------------------------------------------
down() {
  export INFEROPS_PROVIDER=docker-desktop
  unset INFEROPS_KIND_CLUSTER_NAME
  step "Z0. Start of the cleanup"
  run now
  independent_state

  step "Z1. Remove the Application and what it applied, the controller, and the prerequisite layer"
  run bash "${APPLICATION_SCRIPT}" remove --confirm
  run helm --kube-context docker-desktop uninstall inferops --namespace "${NAMESPACE}" --ignore-not-found --wait --timeout 600s
  run bash "${BOOTSTRAP}" remove --confirm
  run bash "${PREREQUISITES}" destroy --confirm

  step "Z2. The cluster after the run, compared with what step P3 read"
  independent_state
  k get namespaces -o name | tr -d '\r' >"${WORK}/namespaces.after"
  k get customresourcedefinitions -o name | tr -d '\r' >"${WORK}/crds.after"
  run diff "${WORK}/namespaces.before" "${WORK}/namespaces.after"
  run diff "${WORK}/crds.before" "${WORK}/crds.after"
  run k get persistentvolumes -o name
  run k get pods --all-namespaces
  run now
}

case "${PHASE}" in
  up) up ;;
  collect) collect ;;
  down) down ;;
  *)
    printf 'REFUSED: the phase is up, collect, or down.\n'
    exit 2
    ;;
esac
