def instant(text: str) -> datetime:
    return datetime.fromisoformat(text.replace("Z", "+00:00"))


def clock(when: datetime) -> str:
    return when.strftime("%H:%M:%S.") + f"{when.microsecond // 1000:03d}Z"


def seconds(ms: float) -> str:
    return f"{ms / 1000:,.1f} s"


def clean(text: str) -> str:
    text = re.sub(r"\x1b\[[0-9;?]*[ -/]*[@-~]", "", text).replace("\r", "")
    windows = str(REPOSITORY)
    posix = REPOSITORY.as_posix()
    forms = {
        windows,
        windows.replace("\\", "\\\\"),
        posix,
        "/" + posix[0].lower() + posix[2:],
        posix[0].lower() + posix[1:],
    }
    for form in sorted(forms, key=len, reverse=True):
        text = text.replace(form, "<repository>")
    text = text.replace("C:/Program Files/Git/usr/bin/bash.exe", "bash")
    text = text.replace("C:\\Program Files\\Git\\usr\\bin\\bash.exe", "bash")
    text = re.sub(
        r"^View build details: docker-desktop://.*$",
        "View build details: <local link removed>",
        text,
        flags=re.MULTILINE,
    )
    return "\n".join(line.rstrip() for line in text.split("\n"))


def jsonl(path: Path) -> list[dict]:
    return [
        json.loads(line)

def offset_midpoint(reading: dict) -> float:
    node = instant(reading["node"])
    low = (node - instant(reading["hostAfter"])).total_seconds() * 1000
    high = (node - instant(reading["hostBefore"])).total_seconds() * 1000
    return (low + high) / 2
